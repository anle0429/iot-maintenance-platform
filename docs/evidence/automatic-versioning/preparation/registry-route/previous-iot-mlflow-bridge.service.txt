# Preserve the existing MLflow server and its loopback-only Docker publication.
[Unit]
Description=Proxy the private K3s socket to existing local MLflow
Requires=iot-mlflow-bridge.socket
After=iot-mlflow-bridge.socket

[Service]
ExecStart=/lib/systemd/systemd-socket-proxyd 127.0.0.1:15000
DynamicUser=yes
NoNewPrivileges=yes
PrivateTmp=yes
ProtectSystem=strict
ProtectHome=yes
MemoryMax=64M
CPUQuota=50%
RestrictAddressFamilies=AF_INET AF_UNIX
