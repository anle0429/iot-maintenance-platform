}

server {
	listen 80;listen [::]:80;
	listen 443 ssl;listen [::]:443 ssl;
	ssl_certificate $secret_dir_path/ssl_keypair_iot-features_gateway-tls.pem;
	ssl_certificate_key $secret_dir_path/ssl_keypair_iot-features_gateway-tls.pem;

	server_tokens off;

	server_name features.iot.test;
	set $resource_type "ingress";
	set $resource_name "features";
	set $resource_namespace "iot-features";
	set $service "-";
	if ($scheme = http) {
		return 308 https://$host:443$request_uri;
	}
	auth_basic "IoT coursework";
	auth_basic_user_file /etc/nginx/secrets/basic_auth_iot-features_gateway-auth;
	location "/" {
		set $service "iot-feature-api";
		proxy_connect_timeout 60s;
		proxy_read_timeout 60s;
		proxy_send_timeout 60s;
		client_max_body_size 1m;
		proxy_set_header Authorization "";
		proxy_set_header Host $host;
		proxy_set_header X-Real-IP $remote_addr;
		proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
		proxy_set_header X-Forwarded-Host $host;
		proxy_set_header X-Forwarded-Port $server_port;
		proxy_set_header X-Forwarded-Proto $scheme;
		proxy_buffering on;
		proxy_pass http://iot-features-features-features.iot.test-iot-feature-api-80;
		
		limit_req zone=iot-features/features burst=5 nodelay;
		
		limit_req_log_level error;
		limit_req_status 429;
		
	}
	
}
