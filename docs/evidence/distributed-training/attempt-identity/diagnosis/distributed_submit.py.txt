"""Submit the existing train kernel from its KFP driver pod to two native Spark Kubernetes executors."""
import argparse,ipaddress,json,os,socket,subprocess
from pathlib import Path
from ml.training_contract import digest,save


def command(workspace,arguments):
    """Build bounded client-mode settings; credentials are read from the pod's token file, never printed."""
    pod=os.environ['HOSTNAME'];address=socket.gethostbyname(socket.gethostname())
    if ipaddress.ip_address(address).is_loopback:raise ValueError('Executors cannot connect to a loopback driver')
    events=Path(workspace)/'spark-events';events.mkdir(exist_ok=False)
    settings={
        'spark.kubernetes.namespace':'kubeflow','spark.kubernetes.driver.pod.name':pod,
        'spark.driver.host':address,'spark.driver.bindAddress':'0.0.0.0','spark.driver.port':'39001','spark.blockManager.port':'39002',
        'spark.kubernetes.authenticate.caCertFile':'/var/run/secrets/kubernetes.io/serviceaccount/ca.crt',
        'spark.kubernetes.authenticate.oauthTokenFile':'/var/run/secrets/kubernetes.io/serviceaccount/token',
        'spark.kubernetes.container.image':'iot-coursework-spark:3.5.9','spark.kubernetes.container.image.pullPolicy':'Never',
        'spark.kubernetes.executor.podTemplateFile':'/workspace/infra/kfp/spark-executor.yaml',
        'spark.kubernetes.executor.podTemplateContainerName':'spark-kubernetes-executor',
        'spark.kubernetes.executor.deleteOnTermination':'false',
        'spark.executor.instances':'2','spark.executor.cores':'1','spark.kubernetes.executor.request.cores':'1',
        'spark.kubernetes.executor.limit.cores':'1','spark.executor.memory':'4g','spark.executor.memoryOverhead':'1g',
        'spark.dynamicAllocation.enabled':'false','spark.speculation':'false',
        'spark.scheduler.minRegisteredResourcesRatio':'1.0','spark.scheduler.maxRegisteredResourcesWaitingTime':'120s',
        'spark.driver.maxResultSize':'128m','spark.eventLog.enabled':'true','spark.eventLog.compress':'false',
        'spark.eventLog.dir':'file://'+str(events),'spark.local.dir':'/ml-scratch',
        'spark.kubernetes.executor.label.iot-coursework-run':Path(workspace).name,
    }
    for name,claim,path,readonly in [('code','iot-ml-code','/workspace','true'),('data','iot-ml-workspace','/ml-data','false'),('spark-local-dir-scratch','iot-ml-scratch','/ml-scratch','false')]:
        prefix='spark.kubernetes.executor.volumes.persistentVolumeClaim.'+name
        settings.update({prefix+'.options.claimName':claim,prefix+'.mount.path':path,prefix+'.mount.readOnly':readonly})
    save(Path(workspace)/'distributed-submit.json',{'passed':True,'driver_pod':pod,'driver_ip':address,'master':'k8s://https://kubernetes.default.svc:443',
        'settings':settings,'code_sha256':digest(Path(__file__)),'scope':'Admission/config only; successful training and task distribution require native runtime/event evidence.'})
    return ['/opt/spark/bin/spark-submit','--master','k8s://https://kubernetes.default.svc:443','--deploy-mode','client','--driver-memory','6g',
        *[part for key,value in settings.items() for part in ['--conf',key+'='+value]],'/workspace/ml/distributed_job.py',*arguments],address


if __name__=='__main__':
    import sys
    arguments=sys.argv[1:];workspace=arguments[arguments.index('--workspace')+1]
    cmd,address=command(workspace,arguments);environment={**os.environ,'SPARK_LOCAL_IP':address}
    result=subprocess.run(cmd,env=environment)
    raise SystemExit(result.returncode)
