def make_backends(tmp_path):
    """Use real drift calculations/storage behind a transport fixture and keep every outgoing payload observable."""
    detector = DriftRuntime(BUNDLE,tmp_path/'drift.db'); observed = []
    def handle(request):
        """Implement only the declared fixed native routes; reject unexpected URLs instead of making network calls."""
        observed.append((str(request.url),request.content))
        if request.url.path=='/health/ready': return httpx2.Response(200,json={'ready':True,'reference_id':REFERENCE})
        if request.url.path=='/v1/drift:compare':
            return httpx2.Response(200,json=detector.compare(CompareRequest.model_validate_json(request.content)))
        identifier = request.url.path.split('/')[3].split(':')[0]
        if identifier=='P-999':return httpx2.Response(404,json={'private_detail':'do not expose this'})
        if request.url.path.endswith(':predict'):
            return httpx2.Response(409 if identifier=='P-006' else 200,json={} if identifier=='P-006' else prediction(identifier))
        return httpx2.Response(200,json=machine(identifier))
    client = httpx2.AsyncClient(transport=httpx2.MockTransport(handle),timeout=2)
    return client,observed,detector

def test_body_guard_counts_stream_bytes_and_preserves_disconnect():
    """A false small length cannot bypass the cap, and a checked replay cannot hide client disconnect."""
    async def scenario(oversized):
        """Drive the ASGI component with explicit body/disconnect events, independent of HTTP-client framing."""
        events=[{'type':'http.request','body':b'x'*65537 if oversized else b'{}','more_body':False},{'type':'http.disconnect'}]
        sent=[];received=[]
        async def receive():
            """Return the original transport's next body or disconnect event."""
            return events.pop(0)
        async def send(event):
            """Retain public HTTP response messages for the boundary assertion."""
            sent.append(event)
        async def app(scope,incoming,outgoing):
            """Read both checked content and the subsequent native disconnect."""
            received.append(await incoming());received.append(await incoming())
        guard=BoundedBearerMiddleware(app,TOKEN)
        await guard({'type':'http','path':'/mcp','method':'POST','headers':[(b'authorization',('Bearer '+TOKEN).encode()),(b'content-length',b'2')]},receive,send)
        if oversized:assert sent[0]['status']==413 and not received
        else:assert received==[{'type':'http.request','body':b'{}','more_body':False},{'type':'http.disconnect'}]
    asyncio.run(scenario(True));asyncio.run(scenario(False))
