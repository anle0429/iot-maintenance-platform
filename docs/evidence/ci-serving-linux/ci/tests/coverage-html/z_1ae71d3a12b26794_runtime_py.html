21def aware(value): 

22 """Reject timestamp ambiguity rather than assuming the server's local timezone.""" 

23 result=datetime.fromisoformat(value.replace('Z','+00:00')) 

24 if result.tzinfo is None:raise Unavailable('Timestamp lacks timezone') 

25 return result.astimezone(timezone.utc)

28class FeatureRuntime: 

29 """Keep eligibility independent of the numeric model and future failure labels.""" 

30 def __init__(self,context,backend,clock=None,model_pin=None): 

31 self.context=context;self.backend=backend;self.clock=clock or (lambda:datetime.now(timezone.utc)) 

32 if context['clock_mode']!='simulation' or not context['machines']:raise Unavailable('Explicit simulation context required') 

33 self.as_of=aware(context['as_of']);self.prepared=aware(context['prepared_wall_utc']) 

34 self.model_pin=validate_pin(LEGACY if model_pin is None else model_pin) 

35 if self.model_pin['protocol_sha256']!=context['protocol_sha256']:raise Unavailable('Feature context protocol differs from the pinned model')

37 def features(self,machine): 

38 """Refuse expired demo lease, unknown context, stale/future features and arrival violations.""" 

39 if not 0<=(self.clock()-self.prepared).total_seconds()<3600:raise Unavailable('Demo lease expired; prepare a fresh separately bound namespace') 

40 if self.context['protocol_sha256']!=self.model_pin['protocol_sha256']:raise Unavailable('Feature context protocol differs from the pinned model') 

41 if machine not in self.context['machines']:raise KeyError(machine) 

42 status=self.context['machines'][machine] 

43 event=aware(status['event_ts']);received=aware(status['received_ts']) 

44 if status['status'] not in ['operating','failed','maintenance'] or not 0<=(self.as_of-event).total_seconds()<=600 or received>self.as_of:raise Unavailable('Operating context is stale or unavailable') 

45 raw=self.backend.fetch(machine) 

46 fields=list(FeatureVector.model_fields) 

47 if set(raw['values'])!=set(fields) or set(raw['timestamps'])!=set(fields):raise Unavailable('Incomplete native feature schema') 

48 timestamps=[aware(raw['timestamps'][n]) for n in fields] 

49 if len(set(timestamps))!=1 or not 0<=(self.as_of-timestamps[0]).total_seconds()<=600:raise Unavailable('Features are stale, inconsistent or from the future') 

50 vector=FeatureVector.model_validate(raw['values']).model_dump() 

51 return {'machine_id':machine,'features':vector,'feature_event_ts':timestamps[0].isoformat(), 

52 'feature_age_seconds':(self.as_of-timestamps[0]).total_seconds(),'operating_context':status, 

53 'as_of':self.as_of.isoformat(),'clock_mode':'simulation','prepared_wall_utc':self.prepared.isoformat(), 

54 'protocol_sha256':self.context['protocol_sha256'],'source_manifest_sha256':self.context['source_manifest_sha256']}

56 def predict(self,machine): 

57 """Call the deployed model only for an operating machine and preserve end-to-end provenance.""" 

58 features=self.features(machine) 

59 if features['operating_context']['status']!='operating':raise NotOperating('Prediction is defined only while the machine is operating') 

60 result=self.backend.predict(features['features']) 

61 metadata=result['model'];predictions=result['predictions'] 

62 if type(metadata.get('production')) is not bool or any(metadata.get(n)!=value for n,value in self.model_pin.items()): 

63 raise Unavailable('KServe model provenance differs') 

64 if len(predictions)!=1:raise Unavailable('Expected exactly one model prediction') 

65 score=predictions[0]['risk_score'];decision=predictions[0]['predicted_failure'] 

66 if isinstance(score,bool) or not isinstance(score,(float,int)) or not 0<=score<=1 or type(decision) is not int or decision!=int(score>=self.model_pin['threshold']):raise Unavailable('Invalid model prediction') 

67 return {'prediction':predictions[0],'model':metadata,'feature_trace':features}

70class FeastKServeBackend:

83 def fetch(self,machine): 

84 """Keep the event timestamp of every field; missing Redis values fail validation upstream.""" 

85 from google.protobuf.json_format import MessageToDict 

86 response=self.store.get_online_features(features=self.service,entity_rows=[{'machine_id':machine}]) 

87 raw=response.to_dict(include_event_timestamps=True) 

88 native=MessageToDict(response.proto,preserving_proto_field_name=True) 

89 vectors=dict(zip(native['metadata']['feature_names']['val'],native['results'],strict=True)) 

90 fields=list(FeatureVector.model_fields) 

91 if raw['machine_id']!=[machine] or any(vectors[n]['statuses']!=['PRESENT'] for n in fields):raise Unavailable('Native feature status is not present') 

92 return {'values':{n:raw[n][0] for n in fields},'timestamps':{n:datetime.fromtimestamp(raw[n+'__ts'][0],timezone.utc).isoformat() for n in fields}}

95 """Bound the downstream wait and avoid caller-controlled URLs, model names or headers.""" 

96 request=urllib.request.Request(self.endpoint+'/v1/models/'+MODEL+':predict',data=json.dumps({'instances':[features]}).encode(),headers={'content-type':'application/json'})
