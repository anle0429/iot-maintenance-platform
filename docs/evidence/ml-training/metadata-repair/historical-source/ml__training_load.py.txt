"""Load all eligible entities through the actual registered Feast service for a KFP dataset artifact."""

import argparse,json
from pathlib import Path
from types import SimpleNamespace

from feature_store.load import run as native_load
from ml.protocol import PROTOCOL_SHA256,load_protocol
from ml.training_contract import digest,inventory,save


def run(args):
    """Run the verified native daily loader into a new workspace and bind all resulting parts."""
    load_protocol(args.protocol)
    destination=args.workspace/'feast-cohort';reports=args.workspace/'feast-load'
    if destination.exists() or reports.exists():raise ValueError('Choose a new KFP workspace')
    native_load(SimpleNamespace(protocol=args.protocol,state=args.state,batch_report=args.batch_report,
        artifacts=args.artifacts,feature_report=args.feature_report,destination=destination,reports=reports))
    report=json.loads((reports/'report.json').read_text())
    if report['native_rows_this_invocation']!=report['total_rows'] or report['total_rows']!=3_132_184:
        raise ValueError('This pipeline load must retrieve the complete declared native cohort, without cached parts')
    save(args.output,{'passed':True,'kind':'feast-cohort','protocol_sha256':PROTOCOL_SHA256,
        'path':str(destination),'report_path':str(reports/'report.json'),'report_sha256':digest(reports/'report.json'),
        'feature_report_sha256':digest(args.feature_report),'split_counts':report['split_counts'],
        'source_manifest_sha256':load_protocol(args.protocol)['source']['manifest_sha256'],
        'artifacts':[{'path':str(destination),'files':inventory(destination)}],
        'code_sha256':digest(Path(__file__))})


if __name__=='__main__':
    parser=argparse.ArgumentParser(description=__doc__)
    for n in ('protocol','state','batch-report','artifacts','feature-report','workspace','output'):
        parser.add_argument('--'+n,type=Path,required=True)
    run(parser.parse_args())
