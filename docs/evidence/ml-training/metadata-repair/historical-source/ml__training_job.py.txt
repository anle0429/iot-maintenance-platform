"""Execute isolated Spark split, train/validation and once-only holdout evaluation KFP steps."""

import argparse,json,sys,time
from datetime import datetime,timezone
from functools import reduce
from pathlib import Path

sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
from ml.protocol import PROTOCOL_SHA256,class_support,load_protocol
from ml.training_contract import digest,inventory,save,select_threshold,verify_descriptor
from pyspark.ml import Pipeline,PipelineModel
from pyspark.ml.classification import LogisticRegression
from pyspark.ml.evaluation import BinaryClassificationEvaluator
from pyspark.ml.feature import StandardScaler,VectorAssembler
from pyspark.sql import SparkSession,functions as F


LABEL='failure_in_next_1h'


def positive_probability(vector):
    """Extract class one without importing the optional pandas-dependent ML helper module."""
    return float(vector[1])


def with_probability(frame):
    """Keep the existing Spark image unchanged while exposing a scalar evaluable probability."""
    return frame.withColumn('risk_score',F.udf(positive_probability,'double')('probability'))


def confusion(frame,threshold):
    """Aggregate all confusion counts in Spark without collecting per-row holdout data."""
    positive=F.col(LABEL)==1;predicted=F.col('risk_score')>=F.lit(threshold)
    values=frame.agg(*[F.sum(F.when(condition,1).otherwise(0)).alias(name) for name,condition in
        [('tp',positive&predicted),('fp',~positive&predicted),('fn',positive&~predicted),('tn',~positive&~predicted)]]).first().asDict()
    values={k:int(v or 0) for k,v in values.items()};tp,fp,fn=values['tp'],values['fp'],values['fn']
    precision=tp/(tp+fp) if tp+fp else 0.0;recall=tp/(tp+fn) if tp+fn else 0.0
    return {**values,'threshold':threshold,'precision':precision,'recall':recall,
        'f1':2*precision*recall/(precision+recall) if precision+recall else 0.0}


def pr_auc(frame):
    """Use explicitly unbinned Spark trapezoidal area under the precision-recall curve."""
    return BinaryClassificationEvaluator(labelCol=LABEL,rawPredictionCol='risk_score',metricName='areaUnderPR',numBins=0).evaluate(frame)


def split(spark,args,protocol):
    """Re-derive immutable temporal partitions and reject changed cohort identity before model access."""
    cohort=verify_descriptor(args.input,'feast-cohort')
    if digest(Path(cohort['report_path']))!=cohort['report_sha256']:raise ValueError('Native Feast report changed')
    frame=spark.read.parquet(str(Path(cohort['path'])/'day=*/data.parquet'))
    fields=protocol['feature_allowlist'];cuts=protocol['chronological_split']
    literal=lambda name:F.lit(cuts[name]).cast('timestamp')
    derived=F.when(F.col('event_timestamp')<literal('train_end_exclusive'),'train').when(F.col('event_timestamp')<literal('validation_end_exclusive'),'validation').otherwise('test')
    upper=F.when(derived=='train',literal('train_end_exclusive')).when(derived=='validation',literal('validation_end_exclusive')).otherwise(literal('observation_end_exclusive'))
    invalid=(F.col('event_timestamp')<literal('start_inclusive'))|(F.col('event_timestamp')>=literal('observation_end_exclusive'))|(F.col('label_available_ts')>=upper)|(F.col('split')!=derived)
    invalid=invalid|~F.col(LABEL).isin(0,1)|F.col(LABEL).isNull()|F.col('machine_id').isNull()|F.col('event_timestamp').isNull()|F.col('label_available_ts').isNull()|F.col('split').isNull()
    for name in fields:invalid=invalid|F.col(name).isNull()|F.isnan(name)|(F.abs(F.col(name))==float('inf'))
    if frame.filter(invalid).limit(1).count():raise ValueError('Null/nonfinite feature, invalid label, time boundary or purge violation')
    if frame.groupBy('machine_id','event_timestamp').count().filter('count<>1').limit(1).count():raise ValueError('Duplicate prediction snapshot')
    summary={}
    rows=frame.groupBy('split').agg(F.count('*').alias('rows'),F.sum(LABEL).alias('positive_rows'),F.countDistinct(F.when(F.col(LABEL)==1,F.col('failure_episode_id'))).alias('failure_episodes'),F.min('event_timestamp').alias('first'),F.max('event_timestamp').alias('last')).collect()
    for row in rows:
        item=row.asDict();name=item.pop('split');item['negative_rows']=item['rows']-item['positive_rows']
        item['first']=item['first'].isoformat();item['last']=item['last'].isoformat();summary[name]=item
    expected={x['split']:(x['rows'],x['positive_rows']) for x in cohort['split_counts']}
    if {n:(v['rows'],v['positive_rows']) for n,v in summary.items()}!=expected or not class_support(summary,protocol)['passed']:
        raise ValueError('Temporal class/episode support differs or fails the predeclared gate')
    target=args.workspace/'splits';artifacts=[]
    if target.exists():raise ValueError('Preserve existing data split')
    for name in ['train','validation','test']:
        destination=target/name
        frame.filter(F.col('split')==name).select('machine_id','event_timestamp','label_available_ts','failure_episode_id',LABEL,*fields).repartition(8).write.mode('errorifexists').parquet(str(destination))
        artifacts.append({'path':str(destination),'files':inventory(destination)})
    save(args.output,{'passed':True,'kind':'temporal-split','protocol_sha256':PROTOCOL_SHA256,
        'input_descriptor_sha256':digest(args.input),'native_report_sha256':cohort['report_sha256'],
        'source_manifest_sha256':cohort['source_manifest_sha256'],'split_paths':{n:str(target/n) for n in summary},
        'counts':summary,'artifacts':artifacts,'code_sha256':digest(Path(__file__))})


def train(spark,args,protocol):
    """Fit scaler/weights/model on train only and seal the validation threshold before holdout access."""
    split_data=verify_descriptor(args.input,'temporal-split');fields=protocol['feature_allowlist']
    if not class_support(split_data['counts'],protocol)['passed']:raise ValueError('Declared class-support gate failed')
    # Only the train and validation Parquet paths are opened in this component.
    training=spark.read.parquet(split_data['split_paths']['train']);validation_source=spark.read.parquet(split_data['split_paths']['validation'])
    counts=split_data['counts']['train'];total,pos=counts['rows'],counts['positive_rows'];prevalence=pos/total
    if not 0<pos<total:raise ValueError('Training data needs both classes')
    weighted=training.withColumn('class_weight',F.when(F.col(LABEL)==1,total/(2*pos)).otherwise(total/(2*(total-pos)))).cache()
    baseline=protocol['baseline']
    estimator=Pipeline(stages=[VectorAssembler(inputCols=fields,outputCol='raw_features',handleInvalid='error'),
        StandardScaler(inputCol='raw_features',outputCol='features',withMean=baseline['standard_scaler_with_mean'],withStd=baseline['standard_scaler_with_std']),
        LogisticRegression(featuresCol='features',labelCol=LABEL,weightCol='class_weight',maxIter=baseline['max_iter'],regParam=baseline['reg_param'])])
    model=estimator.fit(weighted);weighted.unpersist()
    validation=with_probability(model.transform(validation_source)).cache()
    candidates=[confusion(validation,threshold) for threshold in protocol['threshold']['grid']]
    selected=select_threshold(candidates);selected={**selected,'area_under_pr':pr_auc(validation)}
    model.stages[-1].setThreshold(selected['threshold']);output=args.workspace/'trained';output.mkdir(exist_ok=False)
    native=output/'spark-model';model.write().save(str(native))
    scaler,logistic=model.stages[1:]
    portable={'features':fields,'mean':scaler.mean.toArray().tolist(),'std':scaler.std.toArray().tolist(),
        'coefficients':logistic.coefficients.toArray().tolist(),'intercept':float(logistic.intercept),'threshold':selected['threshold'],
        'protocol_sha256':PROTOCOL_SHA256,'kind':'spark-standardscaler-logistic-portable-v1'}
    save(output/'portable.json',portable)
    save(args.output,{'passed':True,'kind':'validation-selected-model','protocol_sha256':PROTOCOL_SHA256,
        'split_descriptor_sha256':digest(args.input),'split_descriptor':str(args.input),'counts':split_data['counts'],
        'model_path':str(native),'portable_path':str(output/'portable.json'),'validation':selected,'validation_candidates':candidates,
        'threshold_selection_partition':'validation','holdout_rows_opened':0,'training_prevalence':prevalence,
        'train_class_weights':{'positive':total/(2*pos),'negative':total/(2*(total-pos))},
        'scaler_fit_partition':'train','spark_model_iterations':int(logistic.summary.totalIterations),
        'artifacts':[{'path':str(output),'files':inventory(output)}],'code_sha256':digest(Path(__file__)),
        'sealed_utc':datetime.now(timezone.utc).isoformat()})
    validation.unpersist()


def evaluate(spark,args,protocol):
    """Open the untouched holdout only after checking the sealed validation selection and model bytes."""
    selected=verify_descriptor(args.input,'validation-selected-model')
    split_path=Path(selected['split_descriptor'])
    if digest(split_path)!=selected['split_descriptor_sha256']:raise ValueError('Data split changed after selection')
    split_data=verify_descriptor(split_path,'temporal-split')
    threshold=selected['validation']['threshold'];portable=json.loads(Path(selected['portable_path']).read_text())
    if selected['holdout_rows_opened']!=0 or portable['threshold']!=threshold or selected['threshold_selection_partition']!='validation':
        raise ValueError('Model/threshold was not sealed before holdout access')
    output=args.workspace/'evaluation';output.mkdir(exist_ok=False)
    save(output/'holdout-opened.json',{'selection_sha256':digest(args.input),'opened_utc':datetime.now(timezone.utc).isoformat(),
        'selected_threshold':threshold,'policy':'One evaluation per fresh workspace; no hyperparameter/threshold changes after test metrics.'})
    model=PipelineModel.load(selected['model_path'])
    predictions=with_probability(model.transform(spark.read.parquet(split_data['split_paths']['test']))).cache()
    test=confusion(predictions,threshold);test['area_under_pr']=pr_auc(predictions)
    comparator=confusion(predictions.withColumn('risk_score',F.lit(selected['training_prevalence'])),threshold)
    comparator['area_under_pr']=pr_auc(predictions.withColumn('risk_score',F.lit(selected['training_prevalence'])))
    terms=[((F.col(n)-F.lit(m))/F.lit(s) if s else F.lit(0.0))*F.lit(c) for n,m,s,c in zip(portable['features'],portable['mean'],portable['std'],portable['coefficients'])]
    z=F.lit(portable['intercept'])+reduce(lambda a,b:a+b,terms)
    score=F.when(z>=0,1/(1+F.exp(-z))).otherwise(F.exp(z)/(1+F.exp(z)))
    discrepancy=predictions.select(F.max(F.abs(F.col('risk_score')-score)).alias('delta')).first()['delta']
    if discrepancy is None or discrepancy>1e-12:raise ValueError('Portable model differs from actual Spark probabilities')
    count=sum(test[x] for x in ['tp','fp','fn','tn'])
    if count!=selected['counts']['test']['rows']:raise ValueError('Holdout row count changed')
    predictions.select('machine_id','event_timestamp',LABEL,'risk_score',*protocol['feature_allowlist']).withColumn('predicted_failure',(F.col('risk_score')>=threshold).cast('int')).write.mode('errorifexists').parquet(str(output/'predictions'))
    save(args.output,{'passed':True,'kind':'holdout-evaluation','protocol_sha256':PROTOCOL_SHA256,
        'selection_descriptor':str(args.input),'selection_sha256':digest(args.input),'fixed_threshold':threshold,
        'validation':selected['validation'],'test':test,'constant_training_prevalence_comparator':comparator,
        'pr_auc_implementation':'Spark3.5.9 BinaryClassificationEvaluator areaUnderPR, numBins=0, trapezoidal PR integral; not average precision',
        'split_counts':selected['counts'],'portable_spark_max_abs_probability_difference':discrepancy,
        'artifacts':[{'path':str(output),'files':inventory(output)}],'code_sha256':digest(Path(__file__)),
        'limitations':protocol['limitations'],'completed_utc':datetime.now(timezone.utc).isoformat()})
    predictions.unpersist()


if __name__=='__main__':
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--step',choices=['split','train','evaluate'],required=True)
    for n in ['input','output','workspace','protocol']:parser.add_argument('--'+n,type=Path,required=True)
    args=parser.parse_args();protocol=load_protocol(args.protocol)
    if args.output.exists():raise ValueError('A completed ML stage cannot be evaluated again in place')
    spark=SparkSession.builder.appName('iot-ml-'+args.step).config('spark.sql.session.timeZone','UTC').config('spark.sql.shuffle.partitions','16').config('spark.sql.adaptive.coalescePartitions.enabled','false').getOrCreate()
    spark.sparkContext.setLogLevel('WARN')
    try:globals()[args.step](spark,args,protocol)
    finally:spark.stop()
